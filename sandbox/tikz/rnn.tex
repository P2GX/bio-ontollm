\documentclass[tikz,border=10pt]{standalone}
\usepackage{amsmath}
\usepackage{amssymb}

\usetikzlibrary{arrows.meta, positioning, calc}

\begin{document}

\begin{tikzpicture}[
    >=Stealth,
    z_node/.style={circle, draw=green!60!black, fill=green!30, minimum size=1.1cm, font=\large},
    h_node/.style={circle, draw=orange!80!black, fill=orange!20, minimum size=1.1cm, font=\large},
    o_node/.style={circle, draw=yellow!80!black, fill=yellow!20, minimum size=1.1cm, font=\large},
    y_node/.style={circle, draw=red!60!black, fill=red!20, minimum size=1.1cm, font=\large},
    loss_node/.style={rectangle, draw=purple!50, fill=purple!10, minimum size=1.1cm, font=\large},
    lbl/.style={font=\small, inner sep=2pt},
    math_lbl/.style={font=\scriptsize, inner sep=3pt}
]

    % Horizontal distance between time steps and from z_t to h_t
    \newcommand{\spacing}{8.8}
    \newcommand{\hstep}{3.4}

    \foreach \i/\sub in {0/{t-1}, 1/{t}, 2/{t+1}} {
        \coordinate (origin) at (\i*\spacing, 0);

        % --- Bottom row ---
        \node[z_node] (z_\i) at (origin) {$\mathbf{z}_{\sub}$};
        \node[h_node] (h_\i) at ($(z_\i) + (\hstep, 0)$) {$\mathbf{h}_{\sub}$};
        \node[lbl]    (x_\i) at ($(z_\i) + (0, -1.8)$) {$\mathbf{x}_{\sub}$};

        % --- Vertical column ---
        \node[o_node]    (o_\i) at ($(h_\i) + (0, 1.8)$) {$\mathbf{o}_{\sub}$};
        \node[y_node]    (y_\i) at ($(o_\i) + (0, 1.8)$) {$\hat{\mathbf{y}}_{\sub}$};
        \node[loss_node] (L_\i) at ($(y_\i) + (0, 1.8)$) {$L_{\sub}$};

        % --- Arrows inside a step ---
        \draw[->] (x_\i) -- node[left, lbl] {$\mathbf{W}_{xh}$} (z_\i);

        \draw[->] (z_\i) -- node[below, math_lbl, yshift=-4mm]
            {$\mathbf{h}_{\sub} = \tanh(\mathbf{z}_{\sub})$} (h_\i);

        \draw[->] (h_\i) -- node[right, lbl] {$\mathbf{W}_{hy}$}
            node[left, math_lbl] {$\mathbf{o}_{\sub} = \mathbf{W}_{hy}\,\mathbf{h}_{\sub} + \mathbf{b}_y$} (o_\i);

        \draw[->] (o_\i) -- node[left, math_lbl]
            {$\hat{\mathbf{y}}_{\sub} = \mathrm{softmax}(\mathbf{o}_{\sub})$} (y_\i);

        \draw[->] (y_\i) -- node[left, math_lbl]
            {$L_{\sub} = -\mathbf{y}_{\sub}^{\top}\log\hat{\mathbf{y}}_{\sub}$} (L_\i);
    }

    % --- Recurrent connections ---
    \draw[->] ($(z_0) - (1.2, 0)$) -- node[above, lbl] {$\mathbf{W}_{hh}$} (z_0);

    \draw[->] (h_0) -- node[below, math_lbl, yshift=-4mm]
        {$\mathbf{z}_t = \mathbf{W}_{hh}\,\mathbf{h}_{t-1} + \mathbf{W}_{xh}\,\mathbf{x}_t + \mathbf{b}_h$}
        node[above, lbl] {$\mathbf{W}_{hh}$} (z_1);

    \draw[->] (h_1) -- node[below, math_lbl, yshift=-4mm]
        {$\mathbf{z}_{t+1} = \mathbf{W}_{hh}\,\mathbf{h}_{t} + \mathbf{W}_{xh}\,\mathbf{x}_{t+1} + \mathbf{b}_h$}
        node[above, lbl] {$\mathbf{W}_{hh}$} (z_2);

    \draw[->] (h_2) -- node[above, lbl] {$\mathbf{W}_{hh}$} ($(h_2) + (1.5, 0)$);

\end{tikzpicture}
\end{document}
